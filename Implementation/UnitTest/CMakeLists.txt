Abstract: The LibEOS unit tests. They link the library objects directly, so that internal classes can be tested.

file(GLOB LIBEOS_UNITTEST_SRC ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)

add_executable(libeos_unittests ${LIBEOS_UNITTEST_SRC} $<TARGET_OBJECTS:libeos_objects>)
set_target_properties(libeos_unittests PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_compile_definitions(libeos_unittests PRIVATE __LIB3MF4EOS_EXPORTS
	LIBEOS_UNITTEST_LIB3MF_LIBRARY="${PROJECT_SOURCE_DIR}/Libraries/lib3mf/bin/lib3mf.dll")
target_include_directories(libeos_unittests PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}
	${CMAKE_CURRENT_AUTOGENERATED_DIR}
	${PROJECT_SOURCE_DIR}/Implementation
	${PROJECT_SOURCE_DIR}
	${PROJECT_SOURCE_DIR}/Libraries/lib3mf/bin/Bindings/CppDynamic)

if (WIN32)
	target_link_libraries(libeos_unittests Shlwapi.lib Winmm.lib Bcrypt.lib Crypt32.lib Version.lib Advapi32.lib)
endif (WIN32)

add_test(NAME libeos_unittests COMMAND libeos_unittests WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
